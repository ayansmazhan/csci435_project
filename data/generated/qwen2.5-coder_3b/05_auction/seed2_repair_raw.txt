Certainly! The errors are due to duplicate variable declarations in the contract. Here's the corrected version:

```solidity
pragma solidity ^0.8.24;

contract Auction {
    address payable public seller;
    uint256 public endTime;
    address public highestBidder;
    uint256 public highestBid;
    mapping(address => uint256) public pendingReturns;

    constructor(uint256 biddingTime) {
        seller = payable(msg.sender);
        endTime = block.timestamp + biddingTime;
    }

    function seller() external view returns (address) {
        return seller;
    }

    function endTime() external view returns (uint256) {
        return endTime;
    }

    function highestBidder() external view returns (address) {
        return highestBidder;
    }

    function highestBid() external view returns (uint256) {
        return highestBid;
    }

    function pendingReturns(address bidder) external view returns (uint256) {
        return pendingReturns[bidder];
    }

    function ended() external view returns (bool) {
        return block.timestamp >= endTime;
    }

    function bid() public payable {
        require(block.timestamp < endTime, "Bidding is closed");
        require(msg.value > highestBid, "Bid must be higher than the current highest bid");

        if (highestBidder != address(0)) {
            pendingReturns[highestBidder] += highestBid;
        }
        highestBidder = msg.sender;
        highestBid = msg.value;
    }

    function withdraw() public {
        require(pendingReturns[msg.sender] > 0, "Nothing to withdraw");
        payable(msg.sender).transfer(pendingReturns[msg.sender]);
        pendingReturns[msg.sender] = 0;
    }

    function end() public {
        require(block.timestamp >= endTime, "Auction has not ended");
        require(highestBidder != address(0), "No bids placed");

        payable(seller).transfer(highestBid);
        highestBidder = address(0);
        highestBid = 0;
    }
}
```

This version removes the duplicate `seller`, `endTime`, `highestBidder`, `highestBid`, and `pendingReturns` variables.
