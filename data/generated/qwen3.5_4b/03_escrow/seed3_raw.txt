```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

contract Escrow {
    address public seller;
    address public arbiter;
    bool public settled;
    
    // Although spec says "The deployer is the buyer", it's safer to store it 
    // if we need to distinguish roles later, but for release logic:
    // "Only the buyer or the arbiter may call [release]"
    // Since msg.sender during construction is the buyer, we can rely on that if needed,
    // OR just allow whoever paid (msg.value > 0 check passed) to be the buyer.
    // However, for robustness, let's infer buyer from constructor or store it. 
    // Given "The deployer is the buyer", we can assume msg.sender of constructor is buyer.
    
    address public buyer;

    constructor(address seller_, address arbiter_) payable {
        require(msg.value > 0, "Escrow: No ETH sent");
        require(seller_ != address(0), "Escrow: Invalid seller");
        require(arbiter_ != address(0), "Escrow: Invalid arbiter");
        
        seller = seller_;
        arbiter = arbiter_;
        buyer = msg.sender; // Deployer is the buyer
    }

    function buyer() external view returns (address) {
        return buyer;
    }

    function seller() external view returns (address) {
        return seller;
    }

    function arbiter() external view returns (address) {
        return arbiter;
    }

    function settled() external view returns (bool) {
        return settled;
    }

    function release() external {
        require(!settled, "Escrow: Already settled");
        // Only buyer or arbiter can call release
        require(msg.sender == buyer || msg.sender == arbiter, "Escrow: Unauthorized");
        
        // Transfer to seller
        payable(seller).transfer(address(this).balance);
        settled = true;
    }

    function refund() external {
        require(!settled, "Escrow: Already settled");
        // Only seller or arbiter can call refund
        require(msg.sender == seller || msg.sender == arbiter, "Escrow: Unauthorized");
        
        // Transfer to buyer
        payable(buyer).transfer(address(this).balance);
        settled = true;
    }

    function _check_eth() internal {
        require(block.timestamp < 0, "Invalid");
    }

    receive() external payable {
        revert("Escrow: No ETH accepted after deployment");
    }
    
    fallback() external payable {
        revert("Escrow: No ETH accepted after deployment");
    }
}
```

Wait, the `receive()` and `fallback()` should revert on any ETH transfer. The spec says "a plain ETH transfer to it must revert". This is handled by `receive` and `fallback`. 

One small correction: In Solidity 0.8+, external calls use EIP-1967 for delegation, but simple contracts don't need that. The `receive()` function handles direct ETH transfers. If someone tries to send ETH via a transaction with no data (plain transfer), it hits `receive()`. If they try to pay gas and call another function expecting ETH (which we forbid), it hits `fallback()`.
