// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

contract Crowdfunding {
    address public creator;
    uint256 public goal;
    uint256 public deadline;
    uint256 public totalRaised;
    mapping(address => uint256) public contributions;
    bool public claimed;

    constructor(uint256 goal_, uint256 duration) {
        require(goal_ > 0, "Goal must be greater than zero");
        creator = msg.sender;
        goal = goal_;
        deadline = block.timestamp + duration;
    }

    function contribute() external payable {
        require(msg.value > 0, "Contribution must be greater than zero");
        require(block.timestamp < deadline, "Campaign has ended");
        
        contributions[msg.sender] += msg.value;
        totalRaised += msg.value;
    }

    function claim() external {
        require(msg.sender == creator, "Only the creator can claim");
        require(block.timestamp >= deadline, "Deadline has not passed");
        require(totalRaised >= goal, "Goal has not been reached");
        require(!claimed, "Campaign has already been claimed");

        (bool success, ) = creator.call{value: totalRaised}("");
        require(success, "Transfer failed");
        
        claimed = true;
    }

    function refund() external {
        require(msg.sender != creator, "Creators cannot refund themselves directly via this logic (they use claim)");
        // Actually, re-reading spec: "every contributor can take their own contribution back"
        // It doesn't explicitly forbid the creator from refunding if they haven't claimed? 
        // However, step 4 says claim happens when goal is reached. 
        // If goal is NOT reached, and it's the creator, should they be able to refund?
        // The spec says "every contributor". Let's allow anyone except we must ensure logic holds.
        // Wait, step 5 conditions: timestamp >= deadline, goal NOT reached, contribution > 0.
        // It doesn't restrict caller to non-creator in the 'unless' block explicitly, 
        // but logically a creator would usually claim if goal met. If goal not met, they are still a contributor (0 or more).
        // Let's allow any caller who meets the criteria.
        
        require(block.timestamp >= deadline, "Deadline has not passed");
        require(totalRaised < goal, "Goal has been reached");
        require(contributions[msg.sender] > 0, "No contribution to refund");

        contributions[msg.sender] = 0;
        (bool success, ) = msg.sender.call{value: contributions[msg.sender]}("");
        // Note: We set it to 0 first in mapping, then send the value. 
        // Actually, we need to calculate the amount before setting to 0.
    }

    // Corrected refund logic to capture amount correctly
    function refund() external {
        require(block.timestamp >= deadline, "Deadline has not passed");
        require(totalRaised < goal, "Goal has been reached");
        
        uint256 contribution = contributions[msg.sender];
        require(contribution > 0, "No contribution to refund");

        (bool success, ) = msg.sender.call{value: contribution}("");
        require(success, "Transfer failed");
        
        contributions[msg.sender] = 0;
    }
    
    // Helper function for clarity if needed, but spec functions are enough.
}
