```solidity
pragma solidity ^0.8.24;

contract Escrow {
    address public buyer;
    address public seller;
    address public arbiter;
    bool public settled;

    constructor(address seller_, address arbiter_) payable {
        if (msg.value == 0) revert("Insufficient ETH sent");
        if (seller_ == address(0) || arbiter_ == address(0)) revert("Zero address provided");
        
        buyer = msg.sender;
        seller = seller_;
        arbiter = arbiter_;
    }

    function buyer() external view returns (address) {
        return buyer;
    }

    function seller() external view returns (address) {
        return seller;
    }

    function arbiter() external view returns (address) {
        return arbiter;
    }

    function settled() external view returns (bool) {
        return settled;
    }

    function release() external {
        require(settled == false, "Escrow is already settled");
        require(msg.sender == buyer || msg.sender == arbiter, "Only buyer or arbiter can release");
        
        payable(seller).transfer(address(this).balance);
        settled = true;
    }

    function refund() external {
        require(settled == false, "Escrow is already settled");
        require(msg.sender == seller || msg.sender == arbiter, "Only seller or arbiter can refund");
        
        payable(buyer).transfer(address(this).balance);
        settled = true;
    }
}
```

This contract implements the required functionality for an escrow contract. It ensures that the deployer is the buyer, that ETH is sent with deployment, and that the constructor reverts if no ETH is sent or if any of the addresses are zero. The `release()` and `refund()` functions send the escrowed amount to the appropriate party and mark the escrow as settled. Only the buyer or arbiter can call these functions, and both functions revert if the escrow is already settled.
