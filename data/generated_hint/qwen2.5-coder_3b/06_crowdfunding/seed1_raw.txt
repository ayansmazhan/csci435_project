```solidity
pragma solidity ^0.8.24;

contract Crowdfunding {
    address public creator;
    uint256 public goal;
    uint256 public deadline;
    uint256 public totalRaised;
    mapping(address => uint256) public contributions;
    bool public claimed;

    constructor(uint256 goal_, uint256 duration) {
        require(goal_ > 0, "Goal must be greater than 0");
        creator = msg.sender;
        goal = goal_;
        deadline = block.timestamp + duration;
    }

    function creator() external view returns (address) {
        return creator;
    }

    function goal() external view returns (uint256) {
        return goal;
    }

    function deadline() external view returns (uint256) {
        return deadline;
    }

    function totalRaised() external view returns (uint256) {
        return totalRaised;
    }

    function contributions(address contributor) external view returns (uint256) {
        return contributions[contributor];
    }

    function claimed() external view returns (bool) {
        return claimed;
    }

    function contribute() external payable {
        require(msg.value > 0, "Contribution amount must be greater than 0");
        require(block.timestamp < deadline, "Deadline has passed");
        contributions[msg.sender] += msg.value;
        totalRaised += msg.value;
    }

    function claim() external {
        require(msg.sender == creator, "Only the creator can claim the funds");
        require(block.timestamp >= deadline, "Deadline has not passed");
        require(goal >= totalRaised, "Goal has not been reached");
        require(!claimed, "Campaign has already been claimed");

        creator.transfer(totalRaised);
        claimed = true;
    }

    function refund() external {
        require(block.timestamp >= deadline, "Deadline has not passed");
        require(goal < totalRaised, "Goal has been reached");
        require(contributions[msg.sender] > 0, "No contribution to refund");

        uint256 amount = contributions[msg.sender];
        contributions[msg.sender] = 0;
        payable(msg.sender).transfer(amount);
    }
}
```

### Explanation:
1. **Constructor**: Initializes the creator, goal, deadline, and totalRaised.
2. **Creator Function**: Returns the address of the creator.
3. **Goal Function**: Returns the set goal amount.
4. **Deadline Function**: Returns the deadline timestamp.
5. **Total Raised Function**: Returns the total amount raised so far.
6. **Contributions Function**: Returns the contribution amount for a specific contributor.
7. **Claimed Function**: Returns whether the campaign has been claimed.
8. **Contribute Function**: Adds the sender's contribution to the total raised amount, ensuring the contribution amount is greater than zero and the deadline has not passed.
9. **Claim Function**: Transfers the total raised amount to the creator if the conditions are met.
10. **Refund Function**: Refunds the sender's contribution if the conditions are met.
