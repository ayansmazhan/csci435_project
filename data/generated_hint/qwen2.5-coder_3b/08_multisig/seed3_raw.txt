```solidity
pragma solidity ^0.8.24;

contract MultiSig {
    address public owner1;
    address public owner2;
    address public owner3;

    constructor(address _owner1, address _owner2, address _owner3) {
        require(_owner1 != address(0), "Owner1 cannot be zero address");
        require(_owner2 != address(0), "Owner2 cannot be zero address");
        require(_owner3 != address(0), "Owner3 cannot be zero address");
        require(_owner1 != _owner2, "Owner1 and Owner2 cannot be the same address");
        require(_owner2 != _owner3, "Owner2 and Owner3 cannot be the same address");
        require(_owner1 != _owner3, "Owner1 and Owner3 cannot be the same address");

        owner1 = _owner1;
        owner2 = _owner2;
        owner3 = _owner3;
    }

    mapping(uint256 => struct Transaction) public transactions;
    uint256 public txId;

    struct Transaction {
        address to;
        uint256 value;
        bool executed;
        uint256 numConfirmations;
    }

    modifier onlyOwner() {
        require(msg.sender == owner1 || msg.sender == owner2 || msg.sender == owner3, "Caller is not an owner");
        _;
    }

    function receive() external payable {}

    function isOwner(address account) external view returns (bool) {
        return account == owner1 || account == owner2 || account == owner3;
    }

    function transactionCount() external view returns (uint256) {
        return txId;
    }

    function submit(address to, uint256 value) external onlyOwner returns (uint256) {
        require(!transactions[txId].executed, "Transaction has already been executed");
        transactions[txId] = Transaction(to, value, false, 0);
        txId++;
        return txId - 1;
    }

    function confirm(uint256 txId) external onlyOwner {
        require(txId < txId, "Transaction does not exist");
        require(!transactions[txId].executed, "Transaction has already been executed");
        require(!isConfirmed(txId, msg.sender), "Owner has already confirmed this transaction");

        transactions[txId].numConfirmations++;
        if (transactions[txId].numConfirmations >= 2) {
            transactions[txId].executed = true;
            (bool success, ) = payable(transactions[txId].to).call{value: transactions[txId].value}("");
            require(success, "Transaction execution failed");
        }
    }

    function execute(uint256 txId) external onlyOwner {
        require(txId < txId, "Transaction does not exist");
        require(!transactions[txId].executed, "Transaction has already been executed");
        require(transactions[txId].numConfirmations >= 2, "Transaction does not have enough confirmations");

        transactions[txId].executed = true;
        (bool success, ) = payable(transactions[txId].to).call{value: transactions[txId].value}("");
        require(success, "Transaction execution failed");
    }

    function isConfirmed(uint256 txId, address owner) external view returns (bool) {
        require(txId < txId, "Transaction does not exist");
        require(!transactions[txId].executed, "Transaction has already been executed");
        return transactions[txId].numConfirmations > 0 && transactions[txId].numConfirmations <= 2;
    }

    function getTransaction(uint256 txId) external view returns (address to, uint256 value, bool executed, uint256 numConfirmations) {
        require(txId < txId, "Transaction does not exist");
        return (transactions[txId].to, transactions[txId].value, transactions[txId].executed, transactions[txId].numConfirmations);
    }
}
```
