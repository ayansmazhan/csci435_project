pragma solidity ^0.8.24;

contract Escrow {
    address public seller;
    address public arbiter;
    bool public settled;

    constructor(address seller_, address arbiter_) payable {
        require(msg.value > 0, "No ETH sent");
        require(seller_ != address(0), "Invalid seller");
        require(arbiter_ != address(0), "Invalid arbiter");
        seller = seller_;
    }

    function release() external {
        require(msg.sender == address(this) || msg.sender == arbiter(), "Unauthorized");
        require(settled == false, "Already settled");
        
        (bool success,) = seller.call{value: msg.value}("");
        require(success, "Transfer failed");
        settled = true;
    }

    function refund() external {
        require(msg.sender == address(this) || msg.sender == seller || msg.sender == arbiter(), "Unauthorized");
        require(settled == false, "Already settled");
        
        (bool success,) = msg.sender.call{value: msg.value}("");
        require(success, "Transfer failed");
        settled = true;
    }

    receive() external {
        revert("No ETH accepted after deployment");
    }
    
    fallback() external {
        revert("No ETH accepted after deployment");
    }
}
