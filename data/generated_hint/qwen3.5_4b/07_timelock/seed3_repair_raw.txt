```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

contract TimeLock {
    mapping(address => uint256) public balances;
    mapping(address => uint256) private _unlockTime;

    function deposit(uint256 lockDuration) external payable {
        require(msg.value > 0, "Deposit value must be greater than zero");
        
        address caller = msg.sender;
        
        if (_unlockTime[caller] == 0 || block.timestamp >= _unlockTime[caller]) {
            _unlockTime[caller] = uint256(block.timestamp) + lockDuration;
        }
        
        balances[caller] += msg.value;
    }

    function increaseLockTime(uint256 extraSeconds) external {
        address caller = msg.sender;
        
        require(balances[caller] > 0, "No balance to increase lock time");
        
        uint256 currentUnlockTime = _unlockTime[caller];
        // Solidity checks for underflow automatically in addition if using uint256, 
        // but we explicitly ensure the result doesn't decrease.
        // Since extraSeconds is unsigned and >= 0, current + extra will always be >= current.
        require(currentUnlockTime + extraSeconds >= currentUnlockTime, "Invalid operation");
        
        _unlockTime[caller] = currentUnlockTime + extraSeconds;
    }

    function withdraw() external {
        address caller = msg.sender;
        
        require(balances[caller] > 0, "No balance to withdraw");
        require(block.timestamp >= _unlockTime[caller], "Vault has not unlocked yet");
        
        (bool success, ) = caller.call{value: balances[caller]}("");
        require(success, "Transfer failed");
        
        balances[caller] = 0;
    }

    function balanceOf(address user) external view returns (uint256) {
        return balances[user];
    }
    
    function unlockTime(address user) external view returns (uint256) {
        return _unlockTime[user];
    }
}
```
