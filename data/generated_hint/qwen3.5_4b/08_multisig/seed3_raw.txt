pragma solidity ^0.8.24;

contract MultiSig {
    address public owner1;
    address public owner2;
    address public owner3;
    
    uint256 public transactionCount;
    
    struct Transaction {
        address to;
        uint256 value;
        bool executed;
        uint256 confirmations;
        mapping(address => bool) confirmedBy;
    }

    Transaction[] transactions;

    constructor(address owner1_, address owner2_, address owner3_) {
        require(owner1_ != address(0), "Owner cannot be zero address");
        require(owner2_ != address(0), "Owner cannot be zero address");
        require(owner3_ != address(0), "Owner cannot be zero address");
        require(owner1_ != owner2_, "Owners must be different");
        require(owner1_ != owner3_, "Owners must be different");
        require(owner2_ != owner3_, "Owners must be different");
        
        owner1 = owner1_;
        owner2 = owner2_;
        owner3 = owner3_;
    }

    receive() external payable {
        // Accept ETH but do not store it in contract balance permanently.
        // The spec implies receiving ETH from users to send via transactions,
        // so we can let the deposit go directly or store it. 
        // Given standard multi-sig patterns where funds are deposited before use:
        // We will assume the user deposits ETH and then submits a transaction.
        // However, since there is no `deposit` function specified, we allow receive() 
        // to just pass through or increase balance if needed. But spec says "accepts plain ETH".
        // To be safe and simple: we don't store it here explicitly unless needed for logic.
        // Since no balance management is required in the interface, we can ignore state change 
        // beyond what's necessary. Let's just accept it to satisfy the requirement.
    }

    function isOwner(address account) external view returns (bool) {
        return account == owner1 || account == owner2 || account == owner3;
    }

    function transactionCount() external view returns (uint256) {
        return transactionCount;
    }

    function submit(address to, uint256 value) external returns (uint256 txId) {
        require(isOwner(msg.sender), "Caller must be an owner");
        
        Transaction storage newTx = transactions[transactionCount];
        newTx.to = to;
        newTx.value = value;
        newTx.executed = false;
        for (uint256 i = 0; i < 3; ++i) {
            require(!newTx.confirmedBy[owner1+i], "Already confirmed by current owner"); // Simplified check below
        }
        
        // Reset confirmations mapping for the new transaction
        newTx.confirmations = 0;
        newTx.confirmedBy[msg.sender] = true;

        transactions[transactionCount] = newTx;
        return transactionCount++;
    }

    function confirm(uint256 txId) external {
        require(txId < transactionCount, "Transaction does not exist");
        Transaction storage tx = transactions[txId];
        
        require(!tx.executed, "Transaction has already been executed");
        require(!tx.confirmedBy[msg.sender], "Already confirmed by current owner");

        tx.confirmations++;
        tx.confirmedBy[msg.sender] = true;
    }

    function execute(uint256 txId) external {
        require(txId < transactionCount, "Transaction does not exist");
        Transaction storage tx = transactions[txId];
        
        require(!tx.executed, "Transaction has already been executed");
        require(tx.confirmations >= 2, "Insufficient confirmations");

        tx.executed = true;
        payable(tx.to).transfer(tx.value);
    }

    function isConfirmed(uint256 txId, address owner) external view returns (bool) {
        require(txId < transactionCount, "Transaction does not exist");
        return transactions[txId].confirmedBy[owner];
    }

    function getTransaction(uint256 txId) external view returns (address to, uint256 value, bool executed, uint256 numConfirmations) {
        require(txId < transactionCount, "Transaction does not exist");
        Transaction storage tx = transactions[txId];
        return (tx.to, tx.value, tx.executed, tx.confirmations);
    }

    // Helper to reset confirmations mapping for a new transaction during submit
    function _resetConfirmations() internal {
        // This is implicitly handled in the constructor loop above but needs to be explicit if logic changes
        // Actually, we handle it inline in submit.
    }
}
